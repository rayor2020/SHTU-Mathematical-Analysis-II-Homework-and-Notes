\subparagraph*{Properties of Fejér Kernel}

\begin{enumerate}
    \item $\frac{1}{2\pi}\int_{-\pi}^{\pi}F_n(t)\,\mathrm{d}t=1$, $n=0,1,2,\dots$
    \item $F_n(t)\geq 0$, $t\in\mathbb{R}$
    \item $\forall\delta>0$, $\int\limits_{\pi\geq|t|\geq\delta}F_n(t)\,\mathrm{d}t\to 0$, $n\to+\infty$
\end{enumerate}

\begin{align*}
    \frac{D_0(t)+D_1(t)+\dots+D_n(t)}{n+1}
     & =\mathcal{F}_n(t)\quad\text{Fejér Kernel}                                                                                           \\
     & =\frac{2\sin\frac{t}{2}}{2^2(n+1)\sin^2\frac{t}{2}}\left[\sin\frac{t}{2}+\sin\frac{3}{2}t+\dots+\sin\frac{2n+1}{2}t\right]          \\

\subparagraph*{Properties of Fourier series}

\begin{theorem}
    Let $f$ be periodic on $\mathbb{R}$ with period $2\pi$, then
    \begin{enumerate}
        \item If $f\in C^k$ (with $k\geq 0$), then $a_n,b_n=\circ\left(\frac{1}{n^k}\right)$
        \item If $f\in C^1$, then $\hat{f'}(n)=in\hat{f}(n)$
        \item If $f$ is $C^0$ and piecewise $C^1$, then $f(x)=\sum_{n\in\mathbb{Z}}\hat{f}(n)\mathrm{e}^{inx}$ \emph{(Fourier inversion formula)}
        \item If $f_h(x)\triangleq f(x+h)$, then $\hat{f_h}(n)=\mathrm{e}^{inh}\hat{f}(n)$
        \item $\widehat{f*g}(n)=2\pi\hat{f}(n)\hat{g}(n)$ where $\widehat{f*g}(x)=\int_{-\pi}^{\pi}f(x-t)g(t)\,\mathrm{d}t$
    \end{enumerate}
\end{theorem}

Proof: $\hat{f'}(n)=\frac{1}{2\pi}\int_{-\pi}^{\pi}f'(t)\mathrm{e}^{-int}\,\mathrm{d}t=in\hat{f}(n)$

\begin{align*}
    \widehat{f*g}(n)
     & =\frac{1}{2\pi}\int_{-\pi}^{\pi}f*g(x)\mathrm{e}^{-inx}\,\mathrm{d}x                                                  \\
     & =\frac{1}{2\pi}\int_{-\pi}^{\pi}\mathrm{e}^{-inx}\,\mathrm{d}x\int_{-\pi}^{\pi}f(x-t)g(t)\,\mathrm{d}t                \\
     & =\frac{1}{2\pi}\int_{-\pi}^{\pi}g(t)\,\mathrm{d}t\int_{-\pi}^{\pi}f(x-t)\mathrm{e}^{-in(x-t)^{-int}}\,\mathrm{d}x     \\
     & =\frac{1}{2\pi}\int_{-\pi}^{\pi}g(t)\mathrm{e}^{-int}\,\mathrm{d}t\int_{-\pi}^{\pi}f(x)\mathrm{e}^{-inx}\,\mathrm{d}x
\end{align*}
