(Riemann-Lebesgue Lemma) Let $\varphi\in R[a,b]$, then $\lim_{\lambda\to+\infty}\int_a^b\varphi(x)\cos \lambda x\,\mathrm{d}x=0$

Proof: $\forall\varepsilon>0$, $\exists T:\,a=t_0<t_1<\dots<t_n=b$ s.t. $\sum_{i=1}^n\omega_i(\varphi,T)\Delta x_i<\varepsilon$

$$\psi\triangleq\begin{cases}
        \varphi(t_i) & x\in[t_i,t_{i+1}),\ i=0,1,2,\dots,n-1 \\
        \varphi(b)   & x=b
    \end{cases}$$

\begin{align*}
    \left|\int_a^b\varphi(x)\cos\lambda x\,\mathrm{d}x\right|
     & =\left|\int_a^b\left(\varphi(x)-\psi(x)\right)\cos\lambda x\,\mathrm{d}x+\int_a^b\psi(x)\cos\lambda x\,\mathrm{d}x\right|                                                                                                                                                                \\
     & \leq\sum_{i=0}^{n-1}\int_{t_i}^{t_{i+1}}\left|\varphi(x)-\psi(x)\right|\,\mathrm{d}x+\underset{=\left|\sum_{i=0}^{n-1}\frac{\varphi(t_i)}{\lambda}(\sin\lambda t_{i+1}-\sin\lambda t_i)\right|}{\left|\sum_{i=0}^{n-1}\int_{t_i}^{t_{i+1}}\varphi(t_i)\cos\lambda x\,\mathrm{d}x\right|} \\
     & \leq\sum_{i=0}^{n-1}\omega_i(\varphi,T)\Delta x_i+\sum_{i=0}^{n-1}\frac{2\left|\varphi(t_i)\right|}{\lambda}
\end{align*}

$\exists\lambda_0=\lambda_0(\varepsilon)>0$ s.t. $\forall\lambda>\lambda_0$, $\left|\int_a^b\varphi(x)\cos\lambda x\,\mathrm{d}x\right|<2\varepsilon$

    \item $f$ period $2\pi$. piecewise $C^1$ (i.e. $\exists t_0<t_1<\dots<t_n$ partition of $[-\pi,\pi]$ s.t. $f|_{(t_i,t_{i+1})}$ is $C^1$, $f(t_i+0),f(t_i-0)$ exists, and $t_+'(t_i)=\lim_{t\to 0^+}\frac{f(t)-f(t_i+0)}{t}$, $f_-'(t_i)$ exists). Then $S_N(f)(x)\to\frac{f(x+0)+f(x-0)}{2}$, $\forall x\in\mathbb{R}$
    \item $f:\,\mathbb{R}\to\mathbb{R}$ has period $2\pi$, $C^0$, and piecewise $C^1$, then $S_N(f)$ uniformly converges to $f$ on $\mathbb{R}$.
\end{enumerate}
