\paragraph{Examples from Signal Processing}

\begin{enumerate}
    \item Half-wave Rectifier: 半波整流函数

          $$f(t)=\begin{cases}
                  0             & -\frac{\pi}{\omega}\leq t\leq 0 \\
                  E\sin\omega t & 0\leq t\leq \frac{\pi}{\omega}
              \end{cases}$$

          \begin{align*}
              a_n
               & =\frac{\omega}{\pi}\int_{-\frac{\pi}{\omega}}^{\frac{\pi}{\omega}}f(t)\cos\left(n\pi\cdot\frac{\omega}{\pi}t\right)\,\mathrm{d}t                                                       \\
               & =\frac{\omega}{\pi}\int_0^{\frac{\pi}{\omega}}E\sin\omega t\cos(n\omega t)\,\mathrm{d}t                                                                                                \\
               & =\frac{E\omega}{2\pi}\int_0^{\frac{\pi}{\omega}}\sin\left((1+n)\omega t\right)+\sin\left((1-n)\omega t\right)\,\mathrm{d}t                                                             \\
               & =\frac{E\omega}{2\pi}\left[\left.\frac{-\cos(n+1)\omega t}{(n+1)\omega}\right|_0^{\frac{\pi}{\omega}}+\left.\frac{\cos(n-1)\omega t}{(n-1)\omega}\right|_0^{\frac{\pi}{\omega}}\right] \\
               & =\frac{E\omega}{2\pi}\left[\frac{(-1)^n+1}{(n+1)\omega}+\frac{(-1)^{n-1}-1}{(n-1)\omega}\right]                                                                                        \\
               & =\frac{E}{2\pi}\left[(-1)^n+1\right]\frac{-2}{n^2-1}                                                                                                                                   \\
               & =\frac{E}{\pi}\frac{(-1)^{n-1}-1}{n^2-1},\quad n\neq 1                                                                                                                                 \\
              a_1
               & =0,\quad n=1                                                                                                                                                                           \\
              b_n
               & =\frac{\omega}{\pi}\int_{-\frac{\pi}{\omega}}^{\frac{\pi}{\omega}}f(t)\sin\left(n\omega t\right)\,\mathrm{d}t                                                                          \\
               & =\frac{\omega}{\pi}\int_0^{\frac{\pi}{\omega}}E\sin\omega t\sin(n\omega t)\,\mathrm{d}t                                                                                                \\
               & =\frac{\omega E}{2\pi}\int_{0}^{\frac{\pi}{\omega}}\left[\cos\left((1+n)\omega t\right)+\cos\left((1-n)\omega t\right)\right]\,\mathrm{d}t                                             \\
               & =\begin{cases}
                      \frac{2 E\omega}{\pi}\left.\left(\frac{\sin(n+1)\omega t}{(n+1)\omega}+\frac{\sin(1-n)\omega t}{(1-n)\omega}\right)\right|_0^{\frac{\pi}{0}}=0 & n\neq 1 \\
                      \frac{E}{2}                                                                                                                                    & n=1
                  \end{cases}
          \end{align*}

          $[-l,l]$, $a_n=\frac{1}{l}\int_{-l}^l f(x)\cos\frac{n\pi}{l}x\,\mathrm{d}x$

          The Fourier Series of $f(t)$ is
          $$\underset{\text{\allbold{direct current}}}{\frac{E}{\pi}}+\frac{E}{2}\underset{\text{\allbold{fundamental waveform}}}{\sin\omega t}+\sum_{n=1}^{\infty}\underset{\text{\allbold{$n$-th harmonic}}}{\frac{E}{\pi}\frac{(-1)^{n-1}-1}{n^2-1}\underset{\boldsymbol{\sin n\omega t}}{\cos n\omega t}}$$
          It converges to $f(t)$ pointwisely.
    \item Rectangular Wave

          Find the Fourier series of $f(x)=\begin{cases}
                  \frac{1}{2h} & |x|\leq h   \\
                  0            & h<|x|\leq l
              \end{cases}$

          \begin{align*}
              a_n
               & =\frac{1}{l}\int_{-l}^l f(x)\cos\left(\frac{n\pi}{l}x\right)\,\mathrm{d}x     \\
               & =\frac{2}{l}\int_0^h\frac{1}{2h}\cos\left(\frac{n\pi}{l}x\right)\,\mathrm{d}x \\
               & =\frac{1}{hl}\frac{l}{n\pi}\left.\sin\left(\frac{n\pi}{l}x\right)\right|_0^h  \\
               & =\frac{\sin\left(\frac{n\pi h}{l}\right)}{n\pi h}                             \\
               & \xlongequal{\text{by Dirichlet Convergence Thm}}\begin{cases}
                                                                     \frac{1}{2h} & |x|<h       \\
                                                                     \frac{1}{4h} & x=\pm h     \\
                                                                     0            & h<|x|\leq l
                                                                 \end{cases}
          \end{align*}

          The Fourier series is
          $$\frac{1}{2}+\frac{1}{\pi}\sum_{n=1}^{\infty}\frac{\sin nh}{nh}\cos nx$$

          $\frac{1}{2l}+\sum_{n=1}^{\infty}\frac{\sin\frac{n\pi h}{l}}{n\pi h}=\frac{1}{2h}$

          $\implies\frac{1}{2}+\sum_{n=1}^{\infty}\frac{\sin nh}{nh}=\frac{\pi}{2h}\xLongrightarrow{h=1}\boxed{\sum_{n=1}^{\infty}\frac{\sin n}{n}=\frac{\pi-1}{2}}$
\end{enumerate}

(Riemann-Lebesgue Lemma) Let $\varphi\in R[a,b]$, then $\lim_{\lambda\to+\infty}\int_a^b\varphi(x)\cos \lambda x\,\mathrm{d}x=0$

Proof: $\forall\varepsilon>0$, $\exists T:\,a=t_0<t_1<\dots<t_n=b$ s.t. $\sum_{i=1}^n\omega_i(\varphi,T)\Delta x_i<\varepsilon$

$$\psi\triangleq\begin{cases}
        \varphi(t_i) & x\in[t_i,t_{i+1}),\ i=0,1,2,\dots,n-1 \\
        \varphi(b)   & x=b
    \end{cases}$$

\begin{align*}
    \left|\int_a^b\varphi(x)\cos\lambda x\,\mathrm{d}x\right|
     & =\left|\int_a^b\left(\varphi(x)-\psi(x)\right)\cos\lambda x\,\mathrm{d}x+\int_a^b\psi(x)\cos\lambda x\,\mathrm{d}x\right|                                                                                                                                                                \\
     & \leq\sum_{i=0}^{n-1}\int_{t_i}^{t_{i+1}}\left|\varphi(x)-\psi(x)\right|\,\mathrm{d}x+\underset{=\left|\sum_{i=0}^{n-1}\frac{\varphi(t_i)}{\lambda}(\sin\lambda t_{i+1}-\sin\lambda t_i)\right|}{\left|\sum_{i=0}^{n-1}\int_{t_i}^{t_{i+1}}\varphi(t_i)\cos\lambda x\,\mathrm{d}x\right|} \\
     & \leq\sum_{i=0}^{n-1}\omega_i(\varphi,T)\Delta x_i+\sum_{i=0}^{n-1}\frac{2\left|\varphi(t_i)\right|}{\lambda}
\end{align*}

$\exists\lambda_0=\lambda_0(\varepsilon)>0$ s.t. $\forall\lambda>\lambda_0$, $\left|\int_a^b\varphi(x)\cos\lambda x\,\mathrm{d}x\right|<2\varepsilon$

(Dirichlet Convergence Thm)

\begin{enumerate}[label=(\Roman*)]
    \item $f$ period $2\pi$. piecewise $C^1$ (i.e. $\exists t_0<t_1<\dots<t_n$ partition of $[-\pi,\pi]$ s.t. $f|_{(t_i,t_{i+1})}$ is $C^1$, $f(t_i+0),f(t_i-0)$ exists, and $t_+'(t_i)=\lim_{t\to 0^+}\frac{f(t)-f(t_i+0)}{t}$, $f_-'(t_i)$ exists). Then $S_N(f)(x)\to\frac{f(x+0)+f(x-0)}{2}$, $\forall x\in\mathbb{R}$
    \item $f:\,\mathbb{R}\to\mathbb{R}$ has period $2\pi$, $C^0$, and piecewise $C^1$, then $S_N(f)$ uniformly converges to $f$ on $\mathbb{R}$.
\end{enumerate}

Proof:

\begin{align*}
    a_n
     & =\frac{1}{\pi}\int_{-\pi}^{\pi}f(x)\cos nx\,\mathrm{d}x                                                                                                  \\
     & \xlongequal{\text{if }C^1}\frac{1}{\pi}\left(\left.\frac{f(x)\sin nx}{n}\right|_{-\pi}^{\pi}-\int_{-\pi}^{\pi}f'(x)\frac{\sin nx}{n}\,\mathrm{d}x\right) \\
     & =-\frac{1}{n}\frac{1}{\pi}\int_{-\pi}^{\pi}f'(x)\sin nx\,\mathrm{d}x                                                                                     \\
    b_n
     & =\frac{1}{\pi}\int_{-\pi}^{\pi}f(x)\sin nx\,\mathrm{d}x                                                                                                  \\
     & =\frac{1}{n}\frac{1}{\pi}\int_{-\pi}^{\pi}f'(x)\cos nx\,\mathrm{d}x
\end{align*}

If $f$ is only piecewise $C^1$ and $f$ is $C^0$, $\left.\frac{f(x)\sin nx}{n}\right|_{t_i}^{t_{i+1}}=\frac{f(t_{i+1})\sin nt_{i+1}-f(t_i)\sin n t_i}{n}$

\begin{align*}
    a_n & =\sum_{i=0}^{n-1}\frac{1}{\pi}\int_{t_i}^{t_{i+1}}f(x)\cos nx\,\mathrm{d}x                           \\
        & =\sum_{i=0}^{n-1}\left(-\frac{1}{n}\frac{1}{\pi}\int_{t_i}^{t_{i+1}}f'(x)\sin nx\,\mathrm{d}x\right) \\
        & =-\frac{1}{n}\frac{1}{\pi}\int_{t_i}^{t_{i+1}}f'(x)\sin nx\,\mathrm{d}x
\end{align*}

$\left.\begin{aligned}
        f  & \leadsto\frac{a_0}{2}+\sum_{n=1}^{\infty}\left(a_n\cos nx+b_n\sin nx\right)     \\
        f' & \leadsto\frac{a_0'}{2}+\sum_{n=1}^{+\infty}\left(a_n'\cos nx+b_n'\sin nx\right)
    \end{aligned}\right>\implies\begin{cases}
        a_n=-\frac{b_n'}{n} \\
        b_n=\frac{a_n'}{n}
    \end{cases}$

\subparagraph*{Bessel's Inequality}

If $f\in\mathbb{R}[-\pi,\pi]$, then
$$\frac{a_0^2}{2}\sum_{n=1}^{\infty}\left(a_n^2+b_n^2\right)\leq\frac{1}{\pi}\int_{-\pi}^{\pi}f^2(x)\,\mathrm{d}x=\|f\|^2_{L^2}$$
where $a_0,a_n,b_n$ are the Fourier coefficients of $f$

(Parsavel, Plancheral Equality)

Proof: $\underset{=\varphi_0}{\frac{1}{\sqrt{2}}},\underset{=\varphi_1}{\cos x},\underset{=\varphi_2}{\sin x},\underset{=\varphi_3}{\cos 2x},\underset{=\varphi_4}{\sin 2x},\dots$ orthonormal,

$$\left<f-\sum_{k=0}^N\left<f,\varphi_k\right>_{L^2}\varphi_k,f-\sum_{k=0}^N\left<f,\varphi_k\right>_{L^2}\varphi_k\right>_{L^2}=\left\|f\right\|_{L^2}^2-2\sum_{k=0}^N\left<f,\varphi_k\right>_{L^2}^2+\sum_{k=0}^N\left<f,\varphi_k\right>_{L^2}^2\geq 0,\ \forall N$$

$\implies\left\|f\right\|_{L^2}^2\geq\sum_{k=0}^N\left<f,\varphi_k\right>_{L^2}^2=\frac{a_0^2}{2}+\left(a_1^2+b_1^2+\dots\right)$

$\left<f,\varphi_0\right>=\frac{1}{\pi}\int_{-\pi}^{\pi}f\cdot\frac{1}{\sqrt{2}}\,\mathrm{d}x=\frac{a_0}{\sqrt{2}}$, $\left<f,\varphi_1\right>=a_1$, $\left<f,\varphi_2\right>=b_1$
