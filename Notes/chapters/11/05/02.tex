\subsubsection[Gauss 定理]{Gauss Theorem}

Suppose $\overrightarrow{F}=P\overrightarrow{i}+Q\overrightarrow{j}+R\overrightarrow{k}$ is a $C^1$ vector field on $\overleftarrow{V}$ (where $V$ ia an bounded open set), and $\partial V$ is a finite disjoint union of piecewise $C^1$ surface, then
$$\oiint\limits_{\left(\partial V,\overrightarrow{n_{\partial V}}\right)}\overrightarrow{F}\cdot\mathrm{d}\overrightarrow{S}=\iiint\limits_V\nabla\cdot\overrightarrow{F}\,\mathrm{d}x\mathrm{d}y\mathrm{d}z$$
where $\overrightarrow{n_{\partial V}}$ is the unit outer normal vector field on $V$ along $\partial V$.

\paragraph*{Gauss's Law for electrostatics}

If $\overrightarrow{r_i}\in V$, $\implies\overrightarrow{E_i}(\overrightarrow{r})=kq_i\cdot\frac{\overrightarrow{r}-\overrightarrow{r_i}}{\left|\overrightarrow{r}-\overrightarrow{r_i}\right|^3}$

\begin{align*}
    \oiint\limits_{\partial V}\overrightarrow{E_i}(\overrightarrow{r})\cdot\mathrm{d}\overrightarrow{S}
     & =\oiint\limits_{\partial V}kq_i\cdot\frac{\overrightarrow{r}-\overrightarrow{r_i}}{\left|\overrightarrow{r}-\overrightarrow{r_i}\right|^3}\cdot\mathrm{d}\overrightarrow{S}                                                                                                                                                                                                                                                                                                  \\
     & =\left(\oiint_{\partial V}-\oiint_{\partial B_{\varepsilon}(\overrightarrow{r_i})}+\oiint_{\partial B_{\varepsilon}(\overrightarrow{r_i})}\right)\overrightarrow{E_i}(\overrightarrow{r})\cdot\mathrm{d}\overrightarrow{S}                                                                                                                                                                                                                                                   \\
     & =\oiint\limits_{\partial(V\setminus B_{\varepsilon}(\overrightarrow{r_i}))}\overrightarrow{E_i}(\overrightarrow{r})\cdot\mathrm{d}\overrightarrow{S}+\oiint\limits_{\partial B_{\varepsilon}(\overrightarrow{r_i})}kq_i\cdot\frac{\overrightarrow{r}-\overrightarrow{r_i}}{\left|\overrightarrow{r}-\overrightarrow{r_i}\right|^3}\cdot\frac{\overrightarrow{r}-\overrightarrow{r_i}}{\left|\overrightarrow{r}-\overrightarrow{r_i}\right|}\cdot\mathrm{d}\overrightarrow{S} \\
     & \xlongequal{\text{Gauss Divergence Theorem}}\iiint\limits_{V\setminus B_{\varepsilon}(\overrightarrow{r_i})}\underset{=0}{\nabla\cdot\overrightarrow{E_i}(\overrightarrow{r})}\,\mathrm{d}V+4\pi kq_i                                                                                                                                                                                                                                                                        \\
     & =4\pi kq_i
\end{align*}

$\frac{\overrightarrow{r}}{\left|\overrightarrow{r}\right|^3}=-\nabla\frac{1}{r}$, $\mathbb{R}^3\setminus\{0\}$

\begin{align*}
    \nabla\cdot\frac{\overrightarrow{r}}{\left|\overrightarrow{r}\right|^3}
     & =-\Delta\frac{1}{r},\quad r=\sqrt{x_1^2+x_2^2+x_3^2}                         \\
     & =-\sum_{i=1}^3\frac{\partial^2}{\partial x_i^2}(r^{-1})                      \\
     & =-\sum_{i=1}^3\frac{\partial}{\partial x_i}\left(-r^{-2}\frac{x_i}{r}\right) \\
     & =\sum_{i=1}^3\frac{\partial}{\partial x_i}(r^{-3}x_i)                        \\
     & =\sum_{i=1}^3(r^{-3}-3r^{-4}\cdot r^{-1}x_i x_i)                             \\
     & =0
\end{align*}

If $\overrightarrow{r_i}\in\overline{V}^C$, then $\oiint\limits_{\partial V}\overrightarrow{E_i}(\overrightarrow{r})\cdot\mathrm{d}\overrightarrow{S}\xlongequal{\text{Gauss Divergence Theorem}}\iiint\limits_{V}\nabla\cdot\overrightarrow{E_i}(\overrightarrow{r})\,\mathrm{d}V=0$
