\paragraph*{Physical meaning of $\nabla f$, $\nabla\times\overrightarrow{F}$, $\nabla\cdot\overrightarrow{F}$}

\begin{itemize}
    \item $\left<\left.\nabla f\right|_A,\overrightarrow{v}\right>=\lim_{L_{\overrightarrow{v}}\to A}\frac{1}{\left|L_{\overrightarrow{v}}\right|}\int_{L_{\overrightarrow{v}}}\,\mathrm{d}F$
    \item $\left<\left.\nabla\times\overrightarrow{F}\right|_A,\overrightarrow{n}\right>=\lim_{S\to A}\frac{1}{\left|\operatorname{Area}(S)\right|}\iint_{S_{\overrightarrow{n}}}\nabla\times\overrightarrow{F}\cdot\mathrm{d}\overrightarrow{S}=\lim_{S\to A}\frac{1}{\left|\operatorname{Area}(S_n)\right|}\oint_{\partial S_n}\overrightarrow{F}\cdot\mathrm{d}\overrightarrow{r}$\
    \item $\left.\nabla\cdot\overrightarrow{F}\right|_A=\lim_{V\to A}\frac{1}{\left|\operatorname{Vol}(V)\right|}\iiint_V\nabla\cdot\overrightarrow{F}\mathrm{d}V=\lim_{V\to A}\frac{1}{\left|\operatorname{Vol}(\Omega)\right|}\oiint_{\partial\Omega}\overrightarrow{F}\cdot\mathrm{d}\overrightarrow{S}$
\end{itemize}

\paragraph*{``Inverse'' operation of $\nabla$, $\nabla\cdot$, $\nabla\times$}

If $\overrightarrow{F}=\nabla f$, then
$$\oint_L\overrightarrow{F}\cdot\mathrm{d}\overrightarrow{r}=\oint_L\nabla f\cdot\mathrm{d}\overrightarrow{r}=0$$
